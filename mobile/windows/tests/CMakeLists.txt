cmake_minimum_required(VERSION 3.14)
project(innocrew_updater_tests LANGUAGES CXX)
enable_testing()
foreach(version IN ITEMS 1 2)
  add_executable(updater_fixture_${version} WIN32 updater_fixture.cpp)
  target_compile_features(updater_fixture_${version} PRIVATE cxx_std_17)
  target_compile_definitions(updater_fixture_${version} PRIVATE FIXTURE_VERSION=${version})
endforeach()
add_executable(desktop_updater_test desktop_updater_test.cpp)
target_compile_features(desktop_updater_test PRIVATE cxx_std_17)
target_compile_options(desktop_updater_test PRIVATE /UNDEBUG /utf-8)
add_test(NAME desktop_updater_policy COMMAND desktop_updater_test)
include(../cmake/winsparkle.cmake)
add_executable(winsparkle_redirect_probe winsparkle_redirect_probe.cpp)
target_link_libraries(winsparkle_redirect_probe PRIVATE WinSparkle)
target_compile_features(winsparkle_redirect_probe PRIVATE cxx_std_17)
add_custom_command(TARGET winsparkle_redirect_probe POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_if_different $<TARGET_FILE:WinSparkle> $<TARGET_FILE_DIR:winsparkle_redirect_probe>)
